export const FileList = ({ files }) => (
    <p className="text-sm text-foreground-subtext">
        {files.length === 0 && 'No file picked yet.'}
        {files.length > 0 && <>Picked: <code className="bg-surface px-1.5 py-0.5 rounded text-xs font-mono text-foreground">{files.map((f) => f.name).join(', ')}</code></>}
    </p>
);

export const DefaultDemo = () => {
    const [files, setFiles] = useState([]);
    return (
        <div className="flex flex-col gap-3">
            <FileInput aria-label="Attachments" multiple onFilesChange={setFiles} />
            <FileList files={files} />
        </div>
    );
};

export const ButtonDemo = () => {

## Usage

```tsx
import { FileInput } from '@aspiralabs/ui';

<FileInput accept="image/*" onFilesChange={(files) => upload(files[0])} />
```

## Opened from a button

Hide the input and open it through a `ref`. The usual pattern for avatar and cover uploads.

```tsx
const ref = useRef<HTMLInputElement>(null);

<Button onClick={() => ref.current?.click()}>Upload avatar</Button>
<FileInput ref={ref} accept="image/*" className="hidden" onFilesChange={handleFiles} />
```

## Over a drop target

Lay a transparent input over a drop area. The browser handles both click-to-pick and drag-and-drop onto the input.

```tsx
<div className="relative ...">
    Click or drop an image here
    <FileInput accept="image/*" className="absolute inset-0 z-10 cursor-pointer opacity-0" onFilesChange={handleFiles} />
</div>
```

## With react-dropzone

Spread `getInputProps()` onto it. The dropzone's `ref`, `style` and handlers all pass through.

```tsx
const { getRootProps, getInputProps } = useDropzone({ onDrop });

<div {...getRootProps()}>
    <FileInput {...getInputProps()} />
    Drop a photo here
</div>
```

## Props

Accepts all native `input` attributes except `type`, `value` and `defaultValue`.

| Prop            | Type                              | Default | Description                                                         |
| --------------- | --------------------------------- | ------- | ------------------------------------------------------------------- |
| `accept`        | `string`                          | —       | Allowed file types, e.g. `image/*`.                                 |
| `multiple`      | `boolean`                         | `false` | Allow picking more than one file.                                   |
| `onChange`      | `(event) => void`                 | —       | Native change event.                                                |
| `onFilesChange` | `(files: File[]) => void`         | —       | Fires after `onChange` with the picked files as an array.           |
| `ref`           | `Ref<HTMLInputElement>`           | —       | The underlying `<input>`. Call `.click()` on it to open the picker. |
| `className`     | `string`                          | —       | Merged last, so `hidden` or an overlay layout overrides the default. |
