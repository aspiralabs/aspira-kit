export const InteractiveSwitch = () => {
    const [on, setOn] = useState(false);
    return (
        <>
            <p className="text-sm text-foreground-subtext">
                Click to toggle. Current: <code className="bg-surface px-1.5 py-0.5 rounded text-xs font-mono text-foreground">{on ? 'true' : 'false'}</code>
            </p>
            <Switch checked={on} onCheckedChange={setOn} />
        </>
    );
};

## Usage

```tsx
import { Switch } from '@aspiralabs/ui';

<Switch checked={enabled} onCheckedChange={setEnabled} />
```

## Props

All props from Radix `Switch.Root` pass through — common ones listed here.

| Prop              | Type                         | Default | Description                                  |
| ----------------- | ---------------------------- | ------- | -------------------------------------------- |
| `checked`         | `boolean`                    | —       | Controlled state.                            |
| `defaultChecked`  | `boolean`                    | `false` | Uncontrolled initial state.                  |
| `onCheckedChange` | `(checked: boolean) => void` | —       | Fires on toggle with the new value.          |
| `disabled`        | `boolean`                    | `false` | Blocks interaction, dims to 50% opacity.     |
| `className`       | `string`                     | —       | Merged onto the root.                        |
